   410→/** Animated horizontal play-count chart — WEB-LAB-8. Bars reflect the simulated, browser-local play counter. */
   411→function TopTracksChart({
   412→  entries,
   413→}: {
   414→  entries: Array<{ track: LabTrack; count: number }>;
   415→}) {
   416→  const max = entries.reduce((m, e) => Math.max(m, e.count), 1);
   417→  const shown = entries.slice(0, 8);
   418→  return (
   419→    <div className="space-y-2" role="img" aria-label={`Bar chart of simulated play counts, top ${shown.length} tracks`}>
   420→      {shown.map((entry, index) => {
   421→        const pct = Math.max(4, Math.round((entry.count / max) * 100));
   422→        return (
   423→          <div key={entry.track.id} className="flex items-center gap-2">
   424→            <span className={`w-5 shrink-0 text-right font-mono text-[11px] ${index === 0 ? 'font-bold text-amber-300' : 'text-muted-foreground'}`}>
   425→              {index + 1}
   426→            </span>
   427→            <div className="min-w-0 flex-1">
   428→              <div className="flex items-baseline justify-between gap-2">
   429→                <span className="truncate text-xs font-medium">{entry.track.title}</span>
   430→                <span className="shrink-0 font-mono text-[10px] text-muted-foreground">
   431→                  {entry.count} {entry.count === 1 ? 'play' : 'plays'}
   432→                </span>
   433→              </div>
   434→              <div className="mt-0.5 h-2 overflow-hidden rounded-full bg-secondary/70">
   435→                <motion.div
   436→                  initial={{ width: 0 }}
   437→                  animate={{ width: `${pct}%` }}
   438→                  transition={{ type: 'spring', stiffness: 120, damping: 20 }}
   439→                  className={`h-full rounded-full ${
   440→                    index === 0
   441→                      ? 'bg-gradient-to-r from-amber-500/80 to-amber-400/60'
   442→                      : 'bg-gradient-to-r from-primary/80 to-primary/50'
   443→                  }`}
   444→                />
   445→              </div>
   446→            </div>
   447→          </div>
   448→        );
   449→      })}
   450→      <p className="pt-1 text-[10px] text-muted-foreground">
   451→        Chart of the SIMULATED play counter (browser-local). The Android app records real play counts in Room.
   452→      </p>
   453→    </div>
   454→  );
   455→}
